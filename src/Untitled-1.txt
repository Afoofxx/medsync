async function handleGenerate() {
  const name = document.getElementById('f-name').value.trim();
  const role = document.getElementById('f-role').value.trim();
  const experience = document.getElementById('f-experience').value.trim();

  if (!name || !role || !experience) {
    alert('Please fill in your name, role, and experience to continue.');
    return;
  }

  goToStep(3);

  document.getElementById('loading-box').style.display = 'flex';
  document.getElementById('output-content').style.display = 'none';
  document.getElementById('btn-copy').style.display = 'none';
  document.getElementById('btn-download').style.display = 'none';
  document.getElementById('btn-regenerate').style.display = 'none';

  const loadingMsgs = [
    'Analysing your experience...',
    'Crafting your narrative...',
    'Optimising for ATS...',
    'Applying finishing touches...'
  ];
  let msgIdx = 0;
  const msgEl = document.getElementById('loading-label');
  const msgInterval = setInterval(() => {
    msgIdx = (msgIdx + 1) % loadingMsgs.length;
    msgEl.textContent = loadingMsgs[msgIdx];
  }, 1800);

  const data = {
    name, role,
    email: document.getElementById('f-email').value.trim(),
    phone: document.getElementById('f-phone').value.trim(),
    location: document.getElementById('f-location').value.trim(),
    linkedin: document.getElementById('f-linkedin').value.trim(),
    experience,
    skills: document.getElementById('f-skills').value.trim(),
    jobdesc: document.getElementById('f-jobdesc').value.trim(),
    degree: document.getElementById('f-degree').value.trim(),
    institution: document.getElementById('f-institution').value.trim()
  };

  try {
    const res = await fetch(GEMINI_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contents: [{ parts: [{ text: buildPrompt(data) }] }],
        generationConfig: { temperature: 0.72, maxOutputTokens: 1400 }
      })
    });

    clearInterval(msgInterval);

    if (!res.ok) {
      const err = await res.json();
      throw new Error(err?.error?.message || 'Gemini API error');
    }

    const result = await res.json();
    let text = result?.candidates?.[0]?.content?.parts?.[0]?.text || 'No content returned.';
    text = text.replace(/\*\*(.*?)\*\*/g, '$1').replace(/\*(.*?)\*/g, '$1').replace(/#{1,4}\s/g, '').trim();

    generatedContent = text;

    document.getElementById('loading-box').style.display = 'none';
    document.getElementById('output-content').style.display = 'block';
    document.getElementById('output-title').textContent = `${selectedDocType} — ${templateNames[selectedTemplate]}`;
    document.getElementById('preview-doc-type').textContent = selectedDocType;
    document.getElementById('preview-template-name').textContent = templateNames[selectedTemplate];
    document.getElementById('doc-text').textContent = text;

    document.getElementById('btn-copy').style.display = 'inline-block';
    document.getElementById('btn-download').style.display = 'flex';
    document.getElementById('btn-regenerate').style.display = 'inline-block';

  } catch (err) {
    clearInterval(msgInterval);
    document.getElementById('loading-box').style.display = 'none';
    document.getElementById('output-content').style.display = 'block';
    document.getElementById('doc-text').textContent = 'Error: ' + err.message + '\n\nPlease go back and try again.';
  }
}
